3   import java.io.FileOutputStream;

5   import java.io.IOException;

8   import org.apache.commons.csv.CSVFormat;

10  
11  import com.google.gson.Gson;
12  import com.google.gson.GsonBuilder;
13  import com.itextpdf.text.Document;
14  import com.itextpdf.text.DocumentException;
15  import com.itextpdf.text.Paragraph;
16  import com.itextpdf.text.pdf.PdfWriter;
17  
18  public class FileHandler {
19          public void saveAsJson(List<Task> tasks, String filePath) throws IOException {
20          Gson gson = new GsonBuilder().setPrettyPrinting().create();
21          try (FileWriter writer = new FileWriter(filePath)) {
22              gson.toJson(tasks, writer);
23          }
24      }
25  
26      public void saveAsCsv(List<Task> tasks, String filePath) throws IOException {
27          try (CSVPrinter printer = new CSVPrinter(new FileWriter(filePath), CSVFormat.DEFAULT.builder().setHeader("Title", "Description", "Completed").build())) {
28              for (Task task : tasks) {
29                  printer.printRecord(task.getTitle(), task.getDescription(), task.isCompleted());
30              }
31          }
32      }
33  
34      public void generatePdf(List<Task> tasks, String filePath) throws DocumentException, IOException {
35          Document document = new Document();
36          PdfWriter.getInstance(document, new FileOutputStream(filePath));
37          document.open();
38          for (Task task : tasks) {
39              document.add(new Paragraph("Title: " + task.getTitle()));
40              document.add(new Paragraph("Description: " + task.getDescription()));
41              document.add(new Paragraph("Completed: " + task.isCompleted()));
42              document.add(new Paragraph("\n"));
43          }
44          document.close();
45      }
46  }
